```python
def process_text(text):
    """
    텍스트 데이터를 분석하고, 감성 분석 수행
    """
    # 텍스트 분석 로직 (감성 분석, 키워드 추출 등)
    sentiment_score = 0.7  # 예시 감성 점수
    # 결과 출력
    print(f"감성 점수: {sentiment_score}")

# 예시

process_text(text)
```
